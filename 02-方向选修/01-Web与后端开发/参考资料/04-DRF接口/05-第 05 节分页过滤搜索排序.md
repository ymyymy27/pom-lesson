## 一、分页（Pagination）

### 1.1 为什么需要分页？

数据库中可能有成千上万条记录，一次性返回所有数据会导致：
- 响应太大，传输慢
- 前端渲染慢
- 内存占用高

### 1.2 全局分页配置

```python
# config/settings.py
REST_FRAMEWORK = {
    'DEFAULT_PAGINATION_CLASS': 'rest_framework.pagination.PageNumberPagination',
    'PAGE_SIZE': 20,
}
```

### 1.3 三种内置分页方式

**1. PageNumberPagination（页码分页）**

```
GET /api/v1/tasks/?page=2&page_size=10

响应：
{
    "count": 50,         // 总数
    "next": "http://.../tasks/?page=3&page_size=10",
    "previous": "http://.../tasks/?page=1&page_size=10",
    "results": [...]     // 当前页数据
}
```

```python
# apps/core/pagination.py
from rest_framework.pagination import PageNumberPagination

class StandardPagination(PageNumberPagination):
    page_size = 20                     # 默认每页条数
    page_size_query_param = 'page_size' # 允许前端指定每页条数
    max_page_size = 100                # 最大每页条数
    page_query_param = 'page'          # 页码参数名
```

**2. LimitOffsetPagination（偏移分页）**

```
GET /api/v1/tasks/?limit=10&offset=20

响应：
{
    "count": 50,
    "next": "http://.../tasks/?limit=10&offset=30",
    "previous": "http://.../tasks/?limit=10&offset=10",
    "results": [...]
}
```

```python
from rest_framework.pagination import LimitOffsetPagination

class StandardLimitOffsetPagination(LimitOffsetPagination):
    default_limit = 20
    max_limit = 100
```

**3. CursorPagination（游标分页，适合实时数据流）**

```python
from rest_framework.pagination import CursorPagination

class TaskCursorPagination(CursorPagination):
    page_size = 20
    ordering = '-created_at'  # 必须指定排序字段
    cursor_query_param = 'cursor'
```

### 1.4 在 ViewSet 中使用

```python
# 全局使用（settings.py 中配置）
REST_FRAMEWORK = {
    'DEFAULT_PAGINATION_CLASS': 'apps.core.pagination.StandardPagination',
    'PAGE_SIZE': 20,
}

# 或在 ViewSet 中指定
class TaskViewSet(viewsets.ModelViewSet):
    queryset = Task.objects.all()
    serializer_class = TaskReadSerializer
    pagination_class = StandardPagination  # 覆盖全局配置

# 禁用分页
class TagViewSet(viewsets.ModelViewSet):
    queryset = Tag.objects.all()
    serializer_class = TagSerializer
    pagination_class = None  # 标签数量少，不需要分页
```

### 1.5 自定义分页响应格式

```python
from rest_framework.pagination import PageNumberPagination
from rest_framework.response import Response

class CustomPagination(PageNumberPagination):
    page_size = 20
    page_size_query_param = 'page_size'
    max_page_size = 100
    
    def get_paginated_response(self, data):
        return Response({
            'pagination': {
                'total': self.page.paginator.count,
                'page': self.page.number,
                'page_size': self.get_page_size(self.request),
                'total_pages': self.page.paginator.num_pages,
                'has_next': self.page.has_next(),
                'has_previous': self.page.has_previous(),
            },
            'results': data,
        })
```

### 2.3 自定义 FilterSet（高级过滤）

```python
# apps/tasks/filters.py
import django_filters
from .models import Task

class TaskFilter(django_filters.FilterSet):
    # 精确匹配
    status = django_filters.ChoiceFilter(choices=Task.Status.choices)
    project = django_filters.NumberFilter(field_name='project_id')
    assignee = django_filters.NumberFilter(field_name='assignee_id')
    
    # 范围过滤
    priority_min = django_filters.NumberFilter(field_name='priority', lookup_expr='gte')
    priority_max = django_filters.NumberFilter(field_name='priority', lookup_expr='lte')
    
    # 日期范围
    created_after = django_filters.DateTimeFilter(field_name='created_at', lookup_expr='gte')
    created_before = django_filters.DateTimeFilter(field_name='created_at', lookup_expr='lte')
    due_before = django_filters.DateFilter(field_name='due_date', lookup_expr='lte')
    
    # 布尔过滤
    is_overdue = django_filters.BooleanFilter(method='filter_overdue')
    unassigned = django_filters.BooleanFilter(method='filter_unassigned')
    
    # 多值过滤
    tags = django_filters.BaseInFilter(field_name='tags__id', lookup_expr='in')
    
    # 字符串包含
    title = django_filters.CharFilter(lookup_expr='icontains')
    
    class Meta:
        model = Task
        fields = ['status', 'priority', 'project', 'assignee']
    
    def filter_overdue(self, queryset, name, value):
        from datetime import date
        if value:
            return queryset.filter(
                due_date__lt=date.today()
            ).exclude(status='completed')
        return queryset
    
    def filter_unassigned(self, queryset, name, value):
        if value:
            return queryset.filter(assignee__isnull=True)
        return queryset.filter(assignee__isnull=False)
```

```python
# apps/tasks/views.py
class TaskViewSet(viewsets.ModelViewSet):
    queryset = Task.objects.all()
    serializer_class = TaskReadSerializer
    filterset_class = TaskFilter  # 使用自定义 FilterSet
    
    # 支持的查询示例：
    # GET /api/v1/tasks/?status=pending
    # GET /api/v1/tasks/?priority_min=5&priority_max=8
    # GET /api/v1/tasks/?created_after=2025-01-01
    # GET /api/v1/tasks/?is_overdue=true
    # GET /api/v1/tasks/?unassigned=true
    # GET /api/v1/tasks/?tags=1,2,3
    # GET /api/v1/tasks/?title=django
```

### 2.4 lookup_expr 速查

| lookup_expr | SQL 对应 | 示例 |
|-------------|---------|------|
| `exact` | `=`（默认） | `status=pending` |
| `iexact` | `ILIKE` | 忽略大小写 |
| `contains` | `LIKE '%x%'` | 包含 |
| `icontains` | `ILIKE '%x%'` | 包含（忽略大小写） |
| `startswith` | `LIKE 'x%'` | 以...开头 |
| `endswith` | `LIKE '%x'` | 以...结尾 |
| `gt` | `>` | 大于 |
| `gte` | `>=` | 大于等于 |
| `lt` | `<` | 小于 |
| `lte` | `<=` | 小于等于 |
| `in` | `IN (...)` | 在列表中 |
| `isnull` | `IS NULL` | 是否为空 |
| `range` | `BETWEEN` | 范围 |

## 三、搜索（Search）

### 3.1 配置搜索

```python
# config/settings.py
REST_FRAMEWORK = {
    'DEFAULT_FILTER_BACKENDS': [
        'django_filters.rest_framework.DjangoFilterBackend',
        'rest_framework.filters.SearchFilter',        # 搜索
        'rest_framework.filters.OrderingFilter',      # 排序
    ],
}
```

```python
class TaskViewSet(viewsets.ModelViewSet):
    queryset = Task.objects.all()
    serializer_class = TaskReadSerializer
    
    # 搜索配置
    search_fields = [
        'title',                  # 精确包含
        'description',            # 精确包含
        '=status',                # 精确匹配（前缀 =）
        '^title',                 # 以...开头（前缀 ^）
        'assignee__username',     # 跨表搜索
        'tags__name',             # 跨表搜索
    ]
    
    # 使用：GET /api/v1/tasks/?search=django
    # 会搜索 title、description 中包含 "django" 的任务
    # 以及 assignee.username、tags.name 匹配的
```

### 3.2 搜索前缀

| 前缀 | 行为 | 示例 |
|------|------|------|
| 无前缀 | `icontains`（包含，忽略大小写） | `'title'` |
| `^` | `istartswith`（以...开头） | `'^title'` |
| `=` | `iexact`（精确匹配） | `'=status'` |
| `@` | 全文搜索（PostgreSQL） | `'@description'` |
| `$` | `iregex`（正则匹配） | `'$title'` |

## 五、完整的 ViewSet 示例

```python
# apps/tasks/views.py
from rest_framework import viewsets, status
from rest_framework.decorators import action
from rest_framework.response import Response
from django_filters.rest_framework import DjangoFilterBackend
from rest_framework.filters import SearchFilter, OrderingFilter

from .models import Task
from .serializers import TaskReadSerializer, TaskWriteSerializer
from .filters import TaskFilter
from apps.core.pagination import CustomPagination

class TaskViewSet(viewsets.ModelViewSet):
    """
    任务管理 API
    
    支持：
    - 分页: ?page=1&page_size=20
    - 过滤: ?status=pending&priority_min=5
    - 搜索: ?search=django
    - 排序: ?ordering=-priority
    """
    pagination_class = CustomPagination
    filterset_class = TaskFilter
    filter_backends = [DjangoFilterBackend, SearchFilter, OrderingFilter]
    search_fields = ['title', 'description', 'assignee__username', 'tags__name']
    ordering_fields = ['created_at', 'updated_at', 'priority', 'due_date']
    ordering = ['-created_at']
    
    def get_queryset(self):
        return Task.objects.select_related(
            'assignee', 'project', 'creator'
        ).prefetch_related('tags').all()
    
    def get_serializer_class(self):
        if self.action in ['list', 'retrieve']:
            return TaskReadSerializer
        return TaskWriteSerializer
    
    def perform_create(self, serializer):
        serializer.save(creator=self.request.user)
    
    @action(detail=False, methods=['get'])
    def statistics(self, request):
        """GET /api/v1/tasks/statistics/"""
        from django.db.models import Count, Q, Avg
        qs = self.filter_queryset(self.get_queryset())  # 应用过滤条件
        stats = qs.aggregate(
            total=Count('id'),
            pending=Count('id', filter=Q(status='pending')),
            in_progress=Count('id', filter=Q(status='in_progress')),
            completed=Count('id', filter=Q(status='completed')),
            cancelled=Count('id', filter=Q(status='cancelled')),
            avg_priority=Avg('priority'),
        )
        return Response(stats)
```
