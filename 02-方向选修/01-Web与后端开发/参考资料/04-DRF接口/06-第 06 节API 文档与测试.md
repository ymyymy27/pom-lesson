# 第 06 节：API 文档与测试

## 一、为什么需要 API 文档？

前后端分离架构中，API 文档是前端开发者和后端开发者之间的 **契约**。好的文档应该包含：
- 每个端点的 URL、HTTP 方法
- 请求参数和请求体格式
- 响应格式和状态码
- 认证方式
- 示例请求和响应

from django.test import TestCase
from django.contrib.auth import get_user_model
from rest_framework.test import APIClient
from rest_framework import status
from apps.tasks.models import Task, Tag
from apps.projects.models import Project

User = get_user_model()

class TaskAPITest(TestCase):
    """任务 API 测试"""
    
    def setUp(self):
        """每个测试方法前执行"""
        self.client = APIClient()
        
        # 创建测试用户
        self.user = User.objects.create_user(
            username='testuser',
            email='test@example.com',
            password='testpass123',

            owner=self.user,

            assignee=self.user,

        self.assertIn('results', response.data)  # 分页响应
    
    def test_list_tasks_filter_by_status(self):
        """测试按状态过滤"""
        response = self.client.get('/api/v1/tasks/', {'status': 'pending'})
        self.assertEqual(response.status_code, status.HTTP_200_OK)
        for task in response.data['results']:
            self.assertEqual(task['status'], 'pending')
    
    def test_list_tasks_search(self):
        """测试搜索"""
        response = self.client.get('/api/v1/tasks/', {'search': '测试'})

    # ===== 创建测试 =====
    
    def test_create_task(self):

        data = {
            'title': '新任务',
            'description': '新任务描述',

            'project': self.project.id,
            'tags': [self.tag.id],

## 五、异常处理

### 5.1 自定义异常处理器

```python
# apps/core/exceptions.py
from rest_framework.views import exception_handler
from rest_framework.response import Response
from rest_framework import status
import logging

logger = logging.getLogger(__name__)

def custom_exception_handler(exc, context):
    """统一 API 错误响应格式"""
    response = exception_handler(exc, context)
    
    if response is not None:
        # DRF 已处理的异常
        error_data = {
            'code': response.status_code,
            'message': _get_error_message(response),
            'errors': response.data if isinstance(response.data, dict) else {'detail': response.data},
        }
        response.data = error_data
    else:
        # DRF 未处理的异常（500 错误）
        logger.error(f"未处理的异常: {exc}", exc_info=True)
        response = Response(
            {
                'code': 500,
                'message': '服务器内部错误',
                'errors': {},
            },
            status=status.HTTP_500_INTERNAL_SERVER_ERROR,
        )
    
    return response

def _get_error_message(response):
    messages = {
        400: '请求参数错误',
        401: '未认证，请先登录',
        403: '没有权限执行此操作',
        404: '请求的资源不存在',
        405: '不支持此 HTTP 方法',
        429: '请求过于频繁',
    }
    return messages.get(response.status_code, '请求错误')
```

```python
# config/settings.py

    'EXCEPTION_HANDLER': 'apps.core.exceptions.custom_exception_handler',
}
```
