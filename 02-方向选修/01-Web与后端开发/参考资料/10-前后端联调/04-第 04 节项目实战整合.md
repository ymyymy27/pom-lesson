## 二、核心页面实现

### 2.1 任务列表页

```tsx
// src/pages/TaskListPage.tsx
import { useState } from 'react';
import { useSearchParams } from 'react-router-dom';
import { useTasks, useCreateTask, useDeleteTask } from '@/hooks/useTasks';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Plus, Search } from 'lucide-react';
import TaskTable from '@/components/TaskTable';
import CreateTaskDialog from '@/components/CreateTaskDialog';
import StatusFilter from '@/components/StatusFilter';
import Pagination from '@/components/Pagination';
import useDebounce from '@/hooks/useDebounce';

export default function TaskListPage() {
    const [searchParams, setSearchParams] = useSearchParams();
    const status = searchParams.get('status') || 'all';
    const page = Number(searchParams.get('page') || '1');
    const [search, setSearch] = useState(searchParams.get('search') || '');
    const debouncedSearch = useDebounce(search, 300);
    
    const { data, isLoading } = useTasks({
        status: status !== 'all' ? status : undefined,
        page,
        search: debouncedSearch || undefined,
    });
    
    const [dialogOpen, setDialogOpen] = useState(false);
    
    return (
        <div className="space-y-4">
            {/* 顶部操作栏 */}
            <div className="flex items-center justify-between">
                <h1 className="text-2xl font-bold">任务列表</h1>
                <Button onClick={() => setDialogOpen(true)}>
                    <Plus className="w-4 h-4 mr-2" /> 新建任务
                </Button>
            </div>
            
            {/* 过滤和搜索 */}
            <div className="flex items-center gap-4">
                <StatusFilter
                    value={status}
                    onChange={(v) => setSearchParams({ status: v, page: '1' })}
                />
                <div className="relative flex-1 max-w-sm">
                    <Search className="absolute left-3 top-2.5 w-4 h-4 text-gray-400" />
                    <Input
                        value={search}
                        onChange={(e) => setSearch(e.target.value)}
                        placeholder="搜索任务..."
                        className="pl-9"
                    />
                </div>
            </div>
            
            {/* 任务表格 */}
            <TaskTable tasks={data?.results ?? []} loading={isLoading} />
            
            {/* 分页 */}
            <Pagination
                page={page}
                total={data?.count ?? 0}
                pageSize={20}
                onChange={(p) => setSearchParams((prev) => { prev.set('page', String(p)); return prev; })}
            />
            
            {/* 新建任务弹窗 */}
            <CreateTaskDialog open={dialogOpen} onOpenChange={setDialogOpen} />
        </div>
    );
}
```

### 2.2 任务详情页

```tsx
// src/pages/TaskDetailPage.tsx
import { useParams, useNavigate } from 'react-router-dom';
import { useTask, useUpdateTask, useDeleteTask } from '@/hooks/useTasks';
import { Button } from '@/components/ui/button';
import { Badge } from '@/components/ui/badge';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import StatusBadge from '@/components/StatusBadge';
import TaskAttachments from '@/components/TaskAttachments';
import Spinner from '@/components/Spinner';

export default function TaskDetailPage() {
    const { id } = useParams<{ id: string }>();
    const navigate = useNavigate();
    const { data: task, isLoading } = useTask(Number(id));
    const updateTask = useUpdateTask();
    const deleteTask = useDeleteTask();
    
    if (isLoading) return <Spinner />;
    if (!task) return <p>任务不存在</p>;
    
    const handleStatusChange = (newStatus: string) => {
        updateTask.mutate({ id: task.id, data: { status: newStatus } });
    };
    
    const handleDelete = () => {
        if (confirm('确定删除这个任务？')) {
            deleteTask.mutate(task.id, {
                onSuccess: () => navigate('/tasks'),
            });
        }
    };
    
    return (
        <div className="max-w-3xl space-y-6">
            {/* 任务头部 */}
            <div className="flex items-start justify-between">
                <div>
                    <h1 className="text-2xl font-bold">{task.title}</h1>
                    <div className="flex items-center gap-3 mt-2">
                        <StatusBadge status={task.status} />
                        <span className="text-sm text-gray-500">
                            优先级: {task.priority}
                        </span>
                        <span className="text-sm text-gray-500">
                            创建者: {task.creator.username}
                        </span>
                    </div>
                </div>
                <div className="flex gap-2">
                    <Button variant="outline" onClick={() => navigate(`/tasks/${task.id}/edit`)}>
                        编辑
                    </Button>
                    <Button variant="destructive" onClick={handleDelete}>删除</Button>
                </div>
            </div>
            
            {/* 任务描述 */}
            <Card>
                <CardHeader><CardTitle>描述</CardTitle></CardHeader>
                <CardContent>
                    <p className="text-gray-700 whitespace-pre-wrap">
                        {task.description || '暂无描述'}
                    </p>
                </CardContent>
            </Card>
            
            {/* 任务信息 */}
            <Card>
                <CardHeader><CardTitle>详细信息</CardTitle></CardHeader>
                <CardContent className="grid grid-cols-2 gap-4 text-sm">
                    <div>
                        <span className="text-gray-500">状态</span>
                        <p><StatusBadge status={task.status} /></p>
                    </div>
                    <div>
                        <span className="text-gray-500">负责人</span>
                        <p>{task.assignee?.username ?? '未分配'}</p>
                    </div>
                    <div>
                        <span className="text-gray-500">项目</span>
                        <p>{task.project.name}</p>
                    </div>
                    <div>
                        <span className="text-gray-500">截止日期</span>
                        <p>{task.dueDate ?? '无'}</p>
                    </div>
                    <div>
                        <span className="text-gray-500">创建时间</span>
                        <p>{new Date(task.createdAt).toLocaleString('zh-CN')}</p>
                    </div>
                    <div>
                        <span className="text-gray-500">更新时间</span>
                        <p>{new Date(task.updatedAt).toLocaleString('zh-CN')}</p>
                    </div>
                </CardContent>
            </Card>
            
            {/* 附件 */}
            <Card>
                <CardHeader><CardTitle>附件</CardTitle></CardHeader>
                <CardContent>
                    <TaskAttachments taskId={task.id} />
                </CardContent>
            </Card>
        </div>
    );
}
```
