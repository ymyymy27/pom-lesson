# 第 01 节：后端性能优化

## 一、数据库查询优化

### 1.1 N+1 问题

N+1 是最常见的性能杀手：查询列表时，每条记录再发一次查询获取关联数据。

```python
# ❌ N+1 问题：1 次查询任务 + N 次查询创建者
tasks = Task.objects.all()
for task in tasks:
    print(task.creator.username)  # 每次循环触发一次 SQL 查询

# ✅ select_related（一对一、外键，使用 JOIN）
tasks = Task.objects.select_related('creator', 'project', 'assignee').all()
for task in tasks:
    print(task.creator.username)  # 不再触发额外查询

# ✅ prefetch_related（多对多、反向外键，使用子查询）
projects = Project.objects.prefetch_related('members', 'tasks').all()
for project in projects:
    print(project.members.count())  # 不再触发额外查询
```

### 1.2 在 DRF 中优化

```python
# apps/tasks/views.py
class TaskViewSet(viewsets.ModelViewSet):
    serializer_class = TaskSerializer
    
    def get_queryset(self):
        return Task.objects.select_related(
            'creator',

            'tags',
            'attachments',
```

### 1.3 只查需要的字段

```python
# ❌ 查出所有字段
tasks = Task.objects.all()

# ✅ values / values_list — 只查指定字段（返回字典/元组）
tasks = Task.objects.values('id', 'title', 'status')

# ✅ only — 延迟加载其他字段（返回 Model 实例）
tasks = Task.objects.only('id', 'title', 'status')

# ✅ defer — 排除某些大字段
tasks = Task.objects.defer('description')  # 暂不加载 description

# 统计查询
from django.db.models import Count, Avg, Q

stats = Task.objects.aggregate(
    total=Count('id'),
    completed=Count('id', filter=Q(status='completed')),
    avg_priority=Avg('priority'),
)
```

## 三、数据库索引

```python
# apps/tasks/models.py
class Task(models.Model):
    title = models.CharField(max_length=200)
    status = models.CharField(max_length=20, db_index=True)  # 单字段索引
    priority = models.IntegerField(default=0)
    created_at = models.DateTimeField(auto_now_add=True)
    project = models.ForeignKey('projects.Project', on_delete=models.CASCADE)
    creator = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE)
    assignee = models.ForeignKey(
        settings.AUTH_USER_MODEL, null=True, on_delete=models.SET_NULL,
        related_name='assigned_tasks',
    )
    
    class Meta:
        indexes = [
            # 复合索引（常用查询组合）
            models.Index(fields=['project', 'status']),
            models.Index(fields=['assignee', 'status']),
            models.Index(fields=['-created_at']),
            models.Index(fields=['-priority', '-created_at']),
        ]
        ordering = ['-created_at']
```

```python
from django.views.decorators.cache import cache_page

@cache_page(60 * 5)  # 缓存 5 分钟
def task_statistics(request):
    ...

# 2. 查询缓存（适合热点查询）
from django.core.cache import cache

def get_project_task_count(project_id):
    key = f'project:{project_id}:task_count'
    count = cache.get(key)
    if count is None:
        count = Task.objects.filter(project_id=project_id).count()
        cache.set(key, count, timeout=300)
    return count

# 3. 序列化器缓存（适合复杂计算的字段）
class ProjectSerializer(serializers.ModelSerializer):
    task_stats = serializers.SerializerMethodField()
    
    def get_task_stats(self, obj):
        key = f'project:{obj.id}:stats'
        stats = cache.get(key)
        if stats is None:
            stats = {
                'total': obj.tasks.count(),
                'completed': obj.tasks.filter(status='completed').count(),
            }
            cache.set(key, stats, timeout=300)
        return stats
```
