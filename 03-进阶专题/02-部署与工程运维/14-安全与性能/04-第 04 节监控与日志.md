```python
import logging

logger = logging.getLogger(__name__)

class TaskViewSet(viewsets.ModelViewSet):
    def perform_create(self, serializer):
        task = serializer.save(creator=self.request.user)
        logger.info(
            'Task created: id=%d, title=%s, user=%s',
            task.id, task.title, self.request.user.username,
        )
    
    def perform_destroy(self, instance):
        logger.warning(
            'Task deleted: id=%d, title=%s, by_user=%s',
            instance.id, instance.title, self.request.user.username,
        )
        instance.delete()

# Celery 任务中
@shared_task
def send_notification_email(user_id, subject, message):
    logger = logging.getLogger('apps.tasks')
    try:
        user = User.objects.get(id=user_id)
        send_mail(subject, message, 'noreply@taskflow.com', [user.email])
        logger.info('Email sent to %s (user_id=%d)', user.email, user_id)
    except User.DoesNotExist:
        logger.error('User not found: user_id=%d', user_id)
    except Exception as e:
        logger.exception('Failed to send email to user_id=%d: %s', user_id, str(e))
```

### 1.3 请求日志中间件

```python
# apps/core/middleware.py
import time
import logging

logger = logging.getLogger('apps.core')

class RequestLogMiddleware:
    def __init__(self, get_response):
        self.get_response = get_response
    
    def __call__(self, request):
        start = time.time()
        
        response = self.get_response(request)
        
        duration = time.time() - start
        
        # 只记录 API 请求
        if request.path.startswith('/api/'):
            user = getattr(request, 'user', None)
            username = user.username if user and user.is_authenticated else 'anonymous'
            
            logger.info(
                '%s %s %d %.3fs user=%s',
                request.method,
                request.get_full_path(),
                response.status_code,
                duration,
                username,
            )
            
            # 慢请求告警
            if duration > 2.0:
                logger.warning(
                    'Slow request: %s %s took %.3fs',
                    request.method, request.get_full_path(), duration,
                )
        
        return response
```

## 四、运行状态监控

### 4.1 健康检查增强版

```python
# apps/core/views.py
from django.http import JsonResponse
from django.db import connection
from django.core.cache import cache
import time

def health_check(request):
    checks = {}
    overall_status = 'ok'
    
    # 数据库检查
    try:
        start = time.time()
        with connection.cursor() as cursor:
            cursor.execute("SELECT 1")
        checks['database'] = {
            'status': 'ok',
            'latency_ms': round((time.time() - start) * 1000, 2),
        }
    except Exception as e:
        checks['database'] = {'status': 'error', 'message': str(e)}
        overall_status = 'unhealthy'
    
    # Redis 检查
    try:
        start = time.time()
        cache.set('_health', '1', timeout=5)
        cache.get('_health')
        checks['redis'] = {
            'status': 'ok',
            'latency_ms': round((time.time() - start) * 1000, 2),
        }
    except Exception as e:
        checks['redis'] = {'status': 'error', 'message': str(e)}
        overall_status = 'unhealthy'
    
    # Celery 检查（简单版）
    try:
        from config.celery import app
        inspector = app.control.inspect()
        active = inspector.active()
        checks['celery'] = {
            'status': 'ok' if active else 'warning',
            'workers': len(active) if active else 0,
        }
    except Exception as e:
        checks['celery'] = {'status': 'error', 'message': str(e)}
    
    http_status = 200 if overall_status == 'ok' else 503
    
    return JsonResponse({
        'status': overall_status,
        'checks': checks,
        'timestamp': time.time(),
    }, status=http_status)
```
