# 视觉模型与图像理解

## 学习目标

- 掌握多模态 LLM 的图像理解能力

## 1. 多模态 AI 概述

```
多模态 = 多种信息形态的融合

文本 ←→ 图像 ←→ 语音 ←→ 视频

主流多模态模型：
- GPT-4o（文本+图像+语音）
- Claude 3.5（文本+图像）
- Gemini 1.5（文本+图像+视频+音频）
- Qwen-VL（文本+图像，开源）
- LLaVA（文本+图像，开源）
```

## 2. 图像理解（Vision API）

### OpenAI GPT-4o

```python
from openai import OpenAI
import base64

client = OpenAI()

# 方式1：URL 图片
response = client.chat.completions.create(
    model="gpt-4o",
    messages=[{
        "role": "user",
        "content": [
            {"type": "text", "text": "描述这张图片的内容"},
            {"type": "image_url", "image_url": {"url": "https://example.com/image.jpg"}},
        ],
    }],
)
print(response.choices[0].message.content)

# 方式2：Base64 本地图片
def encode_image(image_path: str) -> str:
    with open(image_path, "rb") as f:
        return base64.b64encode(f.read()).decode("utf-8")

base64_image = encode_image("photo.jpg")
response = client.chat.completions.create(
    model="gpt-4o",
    messages=[{
        "role": "user",
        "content": [
            {"type": "text", "text": "这张图片中有什么？请详细描述。"},
            {"type": "image_url", "image_url": {
                "url": f"data:image/jpeg;base64,{base64_image}",
                "detail": "high",  # low / high / auto
            }},
        ],
    }],
)

# 多图对比
response = client.chat.completions.create(
    model="gpt-4o",
    messages=[{
        "role": "user",
        "content": [
            {"type": "text", "text": "比较这两张图片的异同"},
            {"type": "image_url", "image_url": {"url": "https://example.com/img1.jpg"}},
            {"type": "image_url", "image_url": {"url": "https://example.com/img2.jpg"}},
        ],
    }],
)
```

### Ollama 视觉模型

```python
import requests
import base64

def ollama_vision(image_path: str, prompt: str, model="llava:7b"):
    with open(image_path, "rb") as f:
        image_b64 = base64.b64encode(f.read()).decode()

    response = requests.post("http://localhost:11434/api/generate", json={
        "model": model,
        "prompt": prompt,
        "images": [image_b64],
        "stream": False,
    })
    return response.json()["response"]

# 使用
result = ollama_vision("screenshot.png", "描述这张截图的内容")
print(result)
```

### 图像分类与标签

```python
def classify_image(image_path: str, categories: list[str]) -> dict:
    """将图片分类到指定类别"""
    base64_image = encode_image(image_path)
    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[{
            "role": "user",
            "content": [
                {"type": "text", "text": f"""将这张图片分类到以下类别之一：{categories}
输出 JSON：{{"category": "类别", "confidence": "高/中/低", "tags": ["标签1", "标签2"]}}"""},
                {"type": "image_url", "image_url": {
                    "url": f"data:image/jpeg;base64,{base64_image}",
                }},
            ],
        }],
        response_format={"type": "json_object"},
    )
    import json
    return json.loads(response.choices[0].message.content)
```

## 4. HuggingFace 视觉模型

```python
from transformers import pipeline

# 图像分类
classifier = pipeline("image-classification", model="google/vit-base-patch16-224")
result = classifier("image.jpg")
print(result)  # [{'label': 'cat', 'score': 0.95}, ...]

# 目标检测
detector = pipeline("object-detection", model="facebook/detr-resnet-50")
result = detector("street.jpg")
for obj in result:
    print(f"{obj['label']}: {obj['score']:.2f} at {obj['box']}")

# 图像描述
captioner = pipeline("image-to-text", model="Salesforce/blip-image-captioning-base")
result = captioner("photo.jpg")
print(result[0]['generated_text'])
```
