# 第 02 节：数据类型与控制流

## 一、基本数据类型

### 1.1 数字类型

```python
# 整数 (int) — 没有大小限制
age = 25
big_number = 10_000_000  # 下划线分隔，提高可读性

# 浮点数 (float)
price = 19.99
pi = 3.14159

# 布尔值 (bool) — True / False，本质是 int 的子类
is_active = True
is_deleted = False

# 算术运算
print(10 + 3)   # 13   加法
print(10 - 3)   # 7    减法
print(10 * 3)   # 30   乘法
print(10 / 3)   # 3.33 除法（返回 float）
print(10 // 3)  # 3    整除
print(10 % 3)   # 1    取余
print(10 ** 3)  # 1000 幂运算
```

### 1.2 字符串 (str)

```python
# 创建字符串
name = "张三"
greeting = '你好'
multi_line = """
这是一个
多行字符串
"""

# 字符串操作
text = "Hello, World"
print(text.upper())        # "HELLO, WORLD"
print(text.lower())        # "hello, world"
print(text.replace("World", "Python"))  # "Hello, Python"
print(text.split(", "))    # ["Hello", "World"]
print(len(text))           # 12
print("World" in text)     # True

# 字符串索引和切片
print(text[0])     # "H"       第一个字符
print(text[-1])    # "d"       最后一个字符
print(text[0:5])   # "Hello"   切片 [起始:结束)
print(text[7:])    # "World"   从第7个到末尾
print(text[:5])    # "Hello"   从开头到第5个

# f-string 格式化（推荐）
name = "李四"
age = 30
print(f"我叫{name}，今年{age}岁")
print(f"明年{age + 1}岁")
print(f"价格：{price:.2f}元")  # 保留两位小数
```

### 1.3 None 类型

```python
result = None  # 表示"没有值"，类似其他语言的 null

if result is None:
    print("结果为空")

# 注意：用 is 而不是 == 来判断 None
```

## 二、容器类型

### 2.1 列表 (list) — 有序、可变

```python
# 创建
fruits = ["苹果", "香蕉", "橘子"]
numbers = [1, 2, 3, 4, 5]
mixed = [1, "hello", True, 3.14]  # 可以混合类型（但不推荐）

# 访问
print(fruits[0])    # "苹果"
print(fruits[-1])   # "橘子"

# 修改
fruits[1] = "葡萄"         # 修改元素
fruits.append("西瓜")      # 末尾添加
fruits.insert(0, "草莓")   # 指定位置插入
fruits.extend(["芒果", "梨"])  # 扩展列表

# 删除
fruits.remove("橘子")      # 按值删除
popped = fruits.pop()       # 弹出最后一个
del fruits[0]               # 按索引删除

# 常用操作
print(len(fruits))          # 长度
print("苹果" in fruits)     # 成员检查
fruits.sort()               # 排序（原地）
sorted_fruits = sorted(fruits)  # 排序（返回新列表）
fruits.reverse()            # 反转

# 列表推导式（非常重要！）
squares = [x ** 2 for x in range(10)]          # [0, 1, 4, 9, 16, ...]
evens = [x for x in range(20) if x % 2 == 0]  # [0, 2, 4, 6, ...]
```

### 2.2 元组 (tuple) — 有序、不可变

```python
# 创建
point = (10, 20)
rgb = (255, 128, 0)
single = (42,)    # 单元素元组必须加逗号

# 访问（和列表一样）
print(point[0])   # 10

# 不可变 — 不能修改、添加、删除
# point[0] = 30   # 报错！TypeError

# 解包
x, y = point
print(x, y)  # 10 20

# 用途：函数返回多个值、字典的键、保护数据不被修改
```

### 2.3 字典 (dict) — 键值对、保持插入顺序（不等于按键排序）

```python
# 创建
user = {
    "name": "张三",
    "age": 25,
    "email": "zhangsan@example.com",
    "is_active": True,
}

# 访问
print(user["name"])           # "张三"
print(user.get("phone"))     # None（键不存在时不报错）
print(user.get("phone", "未填写"))  # "未填写"（提供默认值）

# 修改 & 添加
user["age"] = 26              # 修改
user["phone"] = "13800138000" # 添加新键值对
user.update({"city": "北京", "age": 27})  # 批量更新

# 删除
del user["is_active"]
phone = user.pop("phone")    # 删除并返回值

# 遍历
for key in user:                      # 遍历键
    print(key)

for key, value in user.items():       # 遍历键值对
    print(f"{key}: {value}")

for value in user.values():           # 遍历值
    print(value)

# 字典推导式
squares_dict = {x: x ** 2 for x in range(5)}
# {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}
```

### 2.4 集合 (set) — 无序、不重复

```python
# 创建
tags = {"python", "django", "web"}
numbers = {1, 2, 3, 3, 2, 1}  # 自动去重 → {1, 2, 3}

# 操作
tags.add("api")
tags.remove("web")          # 不存在会报错
tags.discard("web")         # 不存在不报错

# 集合运算
a = {1, 2, 3, 4}
b = {3, 4, 5, 6}
print(a | b)   # 并集 {1, 2, 3, 4, 5, 6}
print(a & b)   # 交集 {3, 4}
print(a - b)   # 差集 {1, 2}

# 常见用途：去重
names = ["张三", "李四", "张三", "王五"]
unique_names = list(set(names))  # ["张三", "李四", "王五"]
```

## 三、类型转换

```python
# 显式类型转换
int("42")        # 42        字符串 → 整数
float("3.14")    # 3.14      字符串 → 浮点数
str(42)          # "42"      整数 → 字符串
list("hello")    # ['h', 'e', 'l', 'l', 'o']
tuple([1, 2, 3]) # (1, 2, 3)
set([1, 2, 2])   # {1, 2}
bool(0)          # False
bool("")         # False
bool([])         # False
bool(None)       # False
bool(1)          # True
bool("hello")    # True

# 查看类型
print(type(42))       # <class 'int'>
print(type("hello"))  # <class 'str'>
print(isinstance(42, int))    # True
print(isinstance("hi", str))  # True
```

## 四、控制流

### 4.1 条件语句

```python
score = 85

# if / elif / else
if score >= 90:
    grade = "优秀"
elif score >= 80:
    grade = "良好"
elif score >= 60:
    grade = "及格"
else:
    grade = "不及格"

print(grade)  # "良好"

# 三元表达式（一行 if-else）
status = "通过" if score >= 60 else "未通过"

# 逻辑运算符
age = 25
has_id = True

if age >= 18 and has_id:        # 与
    print("可以购票")

if age < 12 or age >= 65:       # 或
    print("享受优惠")

if not is_deleted:               # 非
    print("记录有效")
```

### 4.2 循环

```python
# for 循环
for fruit in fruits:
    print(fruit)

# range() — 生成数字序列
for i in range(5):          # 0, 1, 2, 3, 4
    print(i)

for i in range(1, 10, 2):  # 1, 3, 5, 7, 9（起始, 结束, 步长）
    print(i)

# enumerate() — 同时获取索引和值
for index, fruit in enumerate(fruits):
    print(f"{index}: {fruit}")

# zip() — 并行遍历多个序列
names = ["张三", "李四", "王五"]
scores = [90, 85, 78]
for name, score in zip(names, scores):
    print(f"{name}: {score}分")

# while 循环
count = 0
while count < 5:
    print(count)
    count += 1

# break 和 continue
for i in range(10):
    if i == 3:
        continue    # 跳过本次循环
    if i == 7:
        break       # 终止整个循环
    print(i)        # 输出 0, 1, 2, 4, 5, 6
```

### 4.3 match-case（Python 3.10+，类似 switch）

```python
command = "quit"

match command:
    case "start":
        print("启动")
    case "stop":
        print("停止")
    case "quit" | "exit":
        print("退出")
    case _:
        print("未知命令")
```
