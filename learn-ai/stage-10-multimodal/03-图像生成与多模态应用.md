## 1. DALL-E 图像生成

```python
from openai import OpenAI

client = OpenAI()

# 生成图像
response = client.images.generate(
    model="dall-e-3",
    prompt="一只可爱的柴犬坐在樱花树下，水彩画风格",
    size="1024x1024",       # 1024x1024 / 1792x1024 / 1024x1792
    quality="standard",      # standard / hd
    n=1,
)

image_url = response.data[0].url
revised_prompt = response.data[0].revised_prompt  # DALL-E 优化后的提示词
print(f"图片: {image_url}")
print(f"实际提示词: {revised_prompt}")

# 下载保存
import requests
img_data = requests.get(image_url).content
with open("generated.png", "wb") as f:
    f.write(img_data)
```

## 2. Stable Diffusion（本地/开源）

```python
# 使用 diffusers 库
# pip install diffusers transformers accelerate torch
from diffusers import StableDiffusionPipeline
import torch

# 加载模型
pipe = StableDiffusionPipeline.from_pretrained(
    "stabilityai/stable-diffusion-2-1",
    torch_dtype=torch.float16,
)
pipe = pipe.to("cuda")

# 生成
image = pipe(
    prompt="a beautiful sunset over mountains, photorealistic, 4k",
    negative_prompt="blurry, low quality, distorted",
    num_inference_steps=50,
    guidance_scale=7.5,
).images[0]

image.save("sd_output.png")
```

## 3. 图像生成提示词技巧

```
结构化提示词公式：
[主体] + [场景] + [风格] + [细节] + [质量]

示例：
"a young woman reading a book, in a cozy coffee shop,
 warm lighting, oil painting style, highly detailed, 4k resolution"

中文提示词（配合中文模型）：
"一位年轻女性在舒适的咖啡馆里读书，暖色调灯光，
 油画风格，高度细节，4K分辨率"

负面提示词（排除不想要的元素）：
"blurry, low quality, distorted, ugly, watermark, text"
```

## 4. 多模态应用实战

### 图文理解助手

```python
class MultimodalAssistant:
    def __init__(self):
        self.client = OpenAI()
        self.messages = []

    def chat_with_image(self, text: str, image_path: str = None) -> str:
        content = [{"type": "text", "text": text}]

        if image_path:
            import base64
            with open(image_path, "rb") as f:
                b64 = base64.b64encode(f.read()).decode()
            content.append({
                "type": "image_url",
                "image_url": {"url": f"data:image/jpeg;base64,{b64}"}
            })

        self.messages.append({"role": "user", "content": content})
        response = self.client.chat.completions.create(
            model="gpt-4o",
            messages=self.messages,
        )
        reply = response.choices[0].message.content
        self.messages.append({"role": "assistant", "content": reply})
        return reply

assistant = MultimodalAssistant()
# 图片问答
print(assistant.chat_with_image("这张图片里有什么？", "photo.jpg"))
# 追问（记住上下文）
print(assistant.chat_with_image("图片中的物体大概值多少钱？"))
```

### 文档分析流水线

```python
async def document_pipeline(file_path: str, question: str) -> str:
    """多模态文档分析：PDF → 图像 → OCR → LLM"""
    import fitz  # PyMuPDF
    
    doc = fitz.open(file_path)
    all_text = []
    
    for page in doc:
        # 提取文字
        text = page.get_text()
        if text.strip():
            all_text.append(text)
        else:
            # 文字提取失败，用视觉模型 OCR
            pix = page.get_pixmap()
            img_bytes = pix.tobytes("png")
            b64 = base64.b64encode(img_bytes).decode()
            
            response = client.chat.completions.create(
                model="gpt-4o",
                messages=[{
                    "role": "user",
                    "content": [
                        {"type": "text", "text": "提取这页文档中的所有文字"},
                        {"type": "image_url", "image_url": {
                            "url": f"data:image/png;base64,{b64}"
                        }},
                    ],
                }],
            )
            all_text.append(response.choices[0].message.content)
    
    # 用 LLM 回答问题
    context = "\n\n".join(all_text)
    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[
            {"role": "system", "content": f"根据文档内容回答问题：\n\n{context[:8000]}"},
            {"role": "user", "content": question},
        ],
    )
    return response.choices[0].message.content
```
