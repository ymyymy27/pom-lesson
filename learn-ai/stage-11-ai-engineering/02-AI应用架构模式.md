## 2. Prompt 管理

```python
from string import Template
from typing import Optional
import json

class PromptManager:
    """集中管理所有 Prompt 模板"""

    def __init__(self, prompts_dir: str = "./prompts"):
        self.prompts: dict[str, dict] = {}
        self._load_prompts(prompts_dir)

    def _load_prompts(self, dir_path: str):
        import os
        for f in os.listdir(dir_path):
            if f.endswith('.json'):
                with open(os.path.join(dir_path, f)) as fp:
                    data = json.load(fp)
                    self.prompts[data['name']] = data

    def get(self, name: str, **kwargs) -> list[dict]:
        """获取格式化后的消息列表"""
        template = self.prompts[name]
        messages = []
        for msg in template['messages']:
            content = Template(msg['content']).safe_substitute(**kwargs)
            messages.append({"role": msg['role'], "content": content})
        return messages

    def register(self, name: str, messages: list[dict], description: str = ""):
        self.prompts[name] = {
            "name": name,
            "description": description,
            "messages": messages,
        }

# prompts/qa.json 示例：
# {
#   "name": "qa",
#   "description": "知识库问答",
#   "messages": [
#     {"role": "system", "content": "根据上下文回答：\n$context"},
#     {"role": "user", "content": "$question"}
#   ]
# }

pm = PromptManager()
messages = pm.get("qa", context="...", question="什么是RAG？")
```

## 3. LLM 路由

```python
from dataclasses import dataclass
from enum import Enum

class TaskType(Enum):
    CHAT = "chat"
    CODE = "code"
    REASONING = "reasoning"
    TRANSLATION = "translation"
    SUMMARY = "summary"

@dataclass
class ModelConfig:
    name: str
    provider: str          # openai / ollama / deepseek
    base_url: str = None
    cost_per_1k: float = 0  # 每千 token 费用
    max_tokens: int = 4096
    supports_tools: bool = False

class LLMRouter:
    def __init__(self):
        self.models = {
            "gpt-4o": ModelConfig("gpt-4o", "openai", cost_per_1k=0.01, supports_tools=True),
            "gpt-4o-mini": ModelConfig("gpt-4o-mini", "openai", cost_per_1k=0.0003, supports_tools=True),
            "deepseek-chat": ModelConfig("deepseek-chat", "deepseek",
                base_url="https://api.deepseek.com", cost_per_1k=0.0003),
            "qwen2.5:7b": ModelConfig("qwen2.5:7b", "ollama",
                base_url="http://localhost:11434/v1", cost_per_1k=0),
        }

        self.routing_rules = {
            TaskType.CHAT: "gpt-4o-mini",
            TaskType.CODE: "deepseek-chat",
            TaskType.REASONING: "gpt-4o",
            TaskType.TRANSLATION: "qwen2.5:7b",
            TaskType.SUMMARY: "gpt-4o-mini",
        }

    def select_model(self, task_type: TaskType, prefer_local: bool = False) -> ModelConfig:
        if prefer_local:
            return self.models["qwen2.5:7b"]
        model_name = self.routing_rules.get(task_type, "gpt-4o-mini")
        return self.models[model_name]

    def get_client(self, model_config: ModelConfig):
        from openai import AsyncOpenAI
        kwargs = {}
        if model_config.base_url:
            kwargs["base_url"] = model_config.base_url
        if model_config.provider == "ollama":
            kwargs["api_key"] = "ollama"
        return AsyncOpenAI(**kwargs)
```

## 4. 可观测性（Logging & Tracing）

```python
import logging
import time
from functools import wraps
from uuid import uuid4

# 结构化日志
logging.basicConfig(
    format='%(asctime)s %(levelname)s [%(name)s] %(message)s',
    level=logging.INFO
)
logger = logging.getLogger("ai_service")

class AITracer:
    """追踪 AI 调用链路"""

    def __init__(self):
        self.traces = {}

    def start_trace(self, request_id: str = None) -> str:
        trace_id = request_id or str(uuid4())
        self.traces[trace_id] = {
            "start_time": time.time(),
            "steps": [],
        }
        return trace_id

    def add_step(self, trace_id: str, step_name: str, **data):
        self.traces[trace_id]["steps"].append({
            "name": step_name,
            "timestamp": time.time(),
            **data,
        })

    def end_trace(self, trace_id: str) -> dict:
        trace = self.traces[trace_id]
        trace["end_time"] = time.time()
        trace["duration"] = trace["end_time"] - trace["start_time"]
        logger.info(f"Trace {trace_id}: {trace['duration']:.2f}s, {len(trace['steps'])} steps")
        return trace

tracer = AITracer()

# 使用示例
trace_id = tracer.start_trace()
tracer.add_step(trace_id, "embedding", tokens=150)
tracer.add_step(trace_id, "retrieval", num_docs=5)
tracer.add_step(trace_id, "llm_call", model="gpt-4o-mini", tokens=500)
result = tracer.end_trace(trace_id)
```

## 5. 安全与护栏

```python
class ContentGuard:
    """内容安全护栏"""

    def __init__(self):
        self.blocked_patterns = [
            "忽略之前的指令",
            "ignore previous instructions",
            "你现在是",
        ]

    def check_input(self, text: str) -> tuple[bool, str]:
        """检查用户输入是否安全"""
        text_lower = text.lower()

        # Prompt 注入检测
        for pattern in self.blocked_patterns:
            if pattern.lower() in text_lower:
                return False, f"检测到不安全内容: {pattern}"

        # 长度限制
        if len(text) > 10000:
            return False, "输入过长"

        return True, "通过"

    def check_output(self, text: str) -> tuple[bool, str]:
        """检查模型输出是否安全"""
        # 敏感信息检测（如 API Key 泄露）
        import re
        if re.search(r'sk-[a-zA-Z0-9]{20,}', text):
            return False, "输出包含敏感信息"
        return True, "通过"

guard = ContentGuard()
```
