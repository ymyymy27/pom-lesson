## 一、文件操作

### 1.1 基础读写

```python
# 写入文件
with open("output.txt", "w", encoding="utf-8") as f:
    f.write("第一行\n")
    f.write("第二行\n")

# 追加内容
with open("output.txt", "a", encoding="utf-8") as f:
    f.write("追加的内容\n")

# 读取整个文件
with open("output.txt", "r", encoding="utf-8") as f:
    content = f.read()
    print(content)

# 逐行读取
with open("output.txt", "r", encoding="utf-8") as f:
    for line in f:
        print(line.strip())   # strip() 去掉行末换行符

# 读取为列表
with open("output.txt", "r", encoding="utf-8") as f:
    lines = f.readlines()     # ["第一行\n", "第二行\n", ...]
```

### 1.2 文件打开模式

| 模式 | 含义 | 文件不存在时 |
|------|------|-------------|
| `"r"` | 只读（默认） | 报错 |
| `"w"` | 写入（覆盖） | 创建新文件 |
| `"a"` | 追加 | 创建新文件 |
| `"x"` | 创建写入（文件已存在则报错） | 创建新文件 |
| `"rb"` | 二进制读取 | 报错 |
| `"wb"` | 二进制写入 | 创建新文件 |

### 1.3 使用 pathlib（推荐）

```python
from pathlib import Path

# 路径操作
base = Path("data")
file_path = base / "users" / "info.json"    # 拼接路径
print(file_path)          # data/users/info.json
print(file_path.parent)   # data/users
print(file_path.name)     # info.json
print(file_path.stem)     # info
print(file_path.suffix)   # .json

# 创建目录
file_path.parent.mkdir(parents=True, exist_ok=True)

# 检查路径
print(file_path.exists())      # 文件是否存在
print(file_path.is_file())     # 是否是文件
print(file_path.is_dir())      # 是否是目录

# 读写文件
file_path.write_text('{"name": "张三"}', encoding="utf-8")
content = file_path.read_text(encoding="utf-8")

# 遍历目录
for item in Path(".").iterdir():
    print(item)

# 递归查找文件
for py_file in Path(".").rglob("*.py"):
    print(py_file)

# 删除文件
file_path.unlink(missing_ok=True)
```

## 二、JSON 文件操作

JSON 是 Web 开发中最常用的数据交换格式，后端 API 返回的就是 JSON。

### 2.1 基础操作

```python
import json

# Python 对象 → JSON 字符串
data = {
    "name": "张三",
    "age": 25,
    "hobbies": ["编程", "阅读"],
    "address": {
        "city": "北京",
        "zip": "100000"
    },
    "is_active": True,
    "score": None        # Python None → JSON null
}

# 序列化（Python → JSON 字符串）
json_str = json.dumps(data, ensure_ascii=False, indent=2)
print(json_str)

# 反序列化（JSON 字符串 → Python）
parsed = json.loads(json_str)
print(parsed["name"])   # "张三"
```

### 2.2 文件读写

```python
from pathlib import Path

# 写入 JSON 文件
def save_json(filepath: str | Path, data: dict) -> None:
    filepath = Path(filepath)
    filepath.parent.mkdir(parents=True, exist_ok=True)
    with open(filepath, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

# 读取 JSON 文件
def load_json(filepath: str | Path) -> dict:
    with open(filepath, "r", encoding="utf-8") as f:
        return json.load(f)

# 使用
save_json("data/users.json", data)
loaded = load_json("data/users.json")
print(loaded["name"])  # "张三"
```

### 2.3 JSON 与 Python 类型对照

| JSON | Python |
|------|--------|
| `object {}` | `dict` |
| `array []` | `list` |
| `string ""` | `str` |
| `number (int)` | `int` |
| `number (float)` | `float` |
| `true/false` | `True/False` |
| `null` | `None` |

## 三、异常处理

### 3.1 try / except / else / finally

```python
# 基础结构
try:
    # 可能出错的代码
    result = 10 / 0
except ZeroDivisionError:
    # 处理特定异常
    print("不能除以零！")
except (TypeError, ValueError) as e:
    # 处理多种异常
    print(f"类型或值错误：{e}")
except Exception as e:
    # 兜底：处理所有其他异常
    print(f"未知错误：{e}")
else:
    # 没有异常时执行
    print(f"结果是：{result}")
finally:
    # 无论是否异常都执行（通常用于清理资源）
    print("执行完毕")
```

### 3.2 常见异常类型

| 异常 | 触发场景 | 示例 |
|------|---------|------|
| `ValueError` | 值不合法 | `int("abc")` |
| `TypeError` | 类型不匹配 | `"2" + 2` |
| `KeyError` | 字典键不存在 | `d["不存在的键"]` |
| `IndexError` | 列表索引越界 | `lst[100]` |
| `FileNotFoundError` | 文件不存在 | `open("不存在.txt")` |
| `ZeroDivisionError` | 除以零 | `1 / 0` |
| `AttributeError` | 属性不存在 | `"hello".foo()` |
| `ImportError` | 导入失败 | `import 不存在的模块` |
| `PermissionError` | 权限不足 | 写入只读文件 |
| `ConnectionError` | 网络连接失败 | 请求不可达的服务器 |

### 3.3 主动抛出异常

```python
def set_age(age: int) -> None:
    if not isinstance(age, int):
        raise TypeError(f"age 必须是整数，收到 {type(age)}")
    if age < 0 or age > 150:
        raise ValueError(f"age 必须在 0-150 之间，收到 {age}")
    print(f"年龄设置为 {age}")

try:
    set_age(-5)
except ValueError as e:
    print(e)  # "age 必须在 0-150 之间，收到 -5"
```

## 四、日志（logging）

### 4.1 为什么用 logging 而不是 print？

- `print` 只能输出到控制台，无法分级、无法写文件
- `logging` 支持日志级别、文件输出、格式化、多模块统一管理

### 4.2 基础使用

```python
import logging

# 配置
logging.basicConfig(
    level=logging.DEBUG,
    format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
    datefmt="%Y-%m-%d %H:%M:%S",
)

# 获取 logger
logger = logging.getLogger(__name__)

# 5 个级别（从低到高）
logger.debug("调试信息 — 开发时用")
logger.info("一般信息 — 程序正常运行")
logger.warning("警告 — 可能有问题")
logger.error("错误 — 出了问题但程序还能跑")
logger.critical("严重错误 — 程序可能崩溃")
```

### 4.3 输出到文件

```python
import logging
from pathlib import Path

# 确保日志目录存在
Path("logs").mkdir(exist_ok=True)

# 配置
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
    handlers=[
        logging.FileHandler("logs/app.log", encoding="utf-8"),  # 写入文件
        logging.StreamHandler(),                                  # 同时输出到控制台
    ],
)

logger = logging.getLogger("taskflow")
logger.info("应用启动")
```

### 4.4 在异常中使用日志

```python
def process_data(filepath: str) -> dict:
    logger = logging.getLogger(__name__)
    
    try:
        data = load_json(filepath)
        logger.info(f"成功加载数据: {filepath}")
        return data
    except FileNotFoundError:
        logger.error(f"文件不存在: {filepath}")
        raise
    except json.JSONDecodeError as e:
        logger.error(f"JSON 解析失败: {filepath}, 错误: {e}")
        raise
```

## 五、综合示例：JSON 任务管理器

把本节内容串联起来的小项目：

```python
"""简易 JSON 任务管理器 — 综合运用文件操作、异常处理、日志"""

import json
import logging
from pathlib import Path
from datetime import datetime

# 日志配置
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
)
logger = logging.getLogger("task_manager")

# 自定义异常
class TaskNotFoundError(Exception):
    def __init__(self, task_id: int):
        self.task_id = task_id
        super().__init__(f"任务 #{task_id} 不存在")

# 数据文件路径
DATA_FILE = Path("data/tasks.json")

def load_tasks() -> list[dict]:
    """从 JSON 文件加载任务列表"""
    if not DATA_FILE.exists():
        logger.info("数据文件不存在，返回空列表")
        return []
    try:
        content = DATA_FILE.read_text(encoding="utf-8")
        tasks = json.loads(content)
        logger.info(f"加载了 {len(tasks)} 个任务")
        return tasks
    except json.JSONDecodeError as e:
        logger.error(f"JSON 解析失败: {e}")
        return []

def save_tasks(tasks: list[dict]) -> None:
    """保存任务列表到 JSON 文件"""
    DATA_FILE.parent.mkdir(parents=True, exist_ok=True)
    content = json.dumps(tasks, ensure_ascii=False, indent=2)
    DATA_FILE.write_text(content, encoding="utf-8")
    logger.info(f"保存了 {len(tasks)} 个任务")

def add_task(title: str, priority: int = 0) -> dict:
    """添加新任务"""
    if not title.strip():
        raise ValueError("任务标题不能为空")
    if not (0 <= priority <= 10):
        raise ValueError(f"优先级必须在 0-10 之间，收到 {priority}")
    
    tasks = load_tasks()
    new_id = max((t["id"] for t in tasks), default=0) + 1
    
    task = {
        "id": new_id,
        "title": title.strip(),
        "priority": priority,
        "status": "pending",
        "created_at": datetime.now().isoformat(),
    }
    
    tasks.append(task)
    save_tasks(tasks)
    logger.info(f"添加任务: #{new_id} {title}")
    return task

def complete_task(task_id: int) -> dict:
    """完成指定任务"""
    tasks = load_tasks()
    
    for task in tasks:
        if task["id"] == task_id:
            task["status"] = "completed"
            save_tasks(tasks)
            logger.info(f"完成任务: #{task_id}")
            return task
    
    raise TaskNotFoundError(task_id)

# 使用示例
if __name__ == "__main__":
    try:
        add_task("学习 Python 基础", priority=8)
        add_task("搭建开发环境", priority=9)
        add_task("完成第一阶段练习", priority=7)
        
        complete_task(1)
        complete_task(999)  # 触发 TaskNotFoundError
        
    except TaskNotFoundError as e:
        logger.warning(f"操作失败: {e}")
    except ValueError as e:
        logger.warning(f"数据校验失败: {e}")
```
