# 第 03 节：视图与请求响应

## 一、视图是什么？

视图（View）是 Django 中 **处理请求并返回响应** 的核心组件。它接收一个 `HttpRequest` 对象，执行业务逻辑，返回一个 `HttpResponse` 对象。

```
HttpRequest → 视图处理 → HttpResponse
```

Django 有两种视图写法：
- **FBV**（Function-Based View）— 函数视图，简单直接
- **CBV**（Class-Based View）— 类视图，面向对象，可复用

## 二、HttpRequest 对象

当用户发送 HTTP 请求时，Django 会自动创建一个 `HttpRequest` 对象，包含请求的所有信息。

```python
def my_view(request):
    # ===== 请求方法 =====
    request.method          # "GET", "POST", "PUT", "DELETE"
    
    # ===== URL 信息 =====
    request.path            # "/api/tasks/"
    request.get_full_path() # "/api/tasks/?page=2&status=pending"
    
    # ===== 查询参数（GET 参数）=====
    # URL: /api/tasks/?page=2&status=pending
    request.GET                     # QueryDict 对象
    request.GET.get('page', '1')    # '2'（获取单个值，默认 '1'）
    request.GET.get('status')       # 'pending'
    request.GET.getlist('tags')     # ['python', 'django']（多个同名参数）
    
    # ===== 请求体（POST/PUT 数据）=====
    request.POST                    # 表单数据（application/x-www-form-urlencoded）
    request.body                    # 原始请求体（bytes）
    
    # 解析 JSON 请求体
    import json
    data = json.loads(request.body) # 前后端分离常用
    
    # ===== 请求头 =====
    request.headers                 # 字典形式访问
    request.headers.get('Content-Type')
    request.headers.get('Authorization')
    request.META['HTTP_USER_AGENT'] # User-Agent
    
    # ===== 用户信息 =====
    request.user                    # 当前登录用户（需要认证中间件）
    request.user.is_authenticated   # 是否已登录
    
    # ===== 其他 =====
    request.content_type            # 'application/json'
    request.FILES                   # 上传的文件
    request.COOKIES                 # Cookie 字典
    request.session                 # Session 数据
```

## 四、函数视图（FBV）

### 4.1 基础 FBV

```python
# apps/tasks/views.py
import json
from django.http import JsonResponse
from django.views.decorators.http import require_http_methods

@require_http_methods(["GET"])
def task_list(request):
    """获取任务列表"""
    # 后续会从数据库查询，这里先用模拟数据
    tasks = [
        {"id": 1, "title": "学习Django", "status": "pending"},
        {"id": 2, "title": "搭建API", "status": "in_progress"},
    ]
    
    # 支持查询参数过滤
    status = request.GET.get("status")
    if status:
        tasks = [t for t in tasks if t["status"] == status]
    
    return JsonResponse({"data": tasks, "count": len(tasks)})

@require_http_methods(["GET"])
def task_detail(request, pk):
    """获取任务详情"""
    # 模拟数据
    task = {"id": pk, "title": f"任务{pk}", "status": "pending"}
    return JsonResponse(task)
```

### 4.2 处理多种 HTTP 方法

```python
from django.views.decorators.csrf import csrf_exempt
import json

@csrf_exempt  # 前后端分离时暂时禁用 CSRF（后续用 JWT 替代）
def task_view(request, pk=None):
    """根据 HTTP 方法执行不同操作"""
    
    if request.method == "GET":
        if pk:
            # GET /api/tasks/5/ → 获取详情
            return JsonResponse({"id": pk, "title": f"任务{pk}"})
        else:
            # GET /api/tasks/ → 获取列表
            return JsonResponse({"data": [], "count": 0})
    
    elif request.method == "POST":
        # POST /api/tasks/ → 创建任务
        data = json.loads(request.body)
        return JsonResponse({"id": 1, **data}, status=201)
    
    elif request.method == "PUT":
        # PUT /api/tasks/5/ → 更新任务
        data = json.loads(request.body)
        return JsonResponse({"id": pk, **data})
    
    elif request.method == "DELETE":
        # DELETE /api/tasks/5/ → 删除任务
        return JsonResponse({"message": f"任务{pk}已删除"}, status=204)
    
    return JsonResponse({"error": "不支持的方法"}, status=405)
```

## 五、类视图（CBV）

### 5.1 基础 CBV

类视图将不同的 HTTP 方法映射到类的方法上，代码更整洁。

```python
from django.views import View
from django.http import JsonResponse
from django.views.decorators.csrf import csrf_exempt
from django.utils.decorators import method_decorator
import json

@method_decorator(csrf_exempt, name='dispatch')
class TaskListView(View):
    """任务列表视图"""
    
    def get(self, request):
        """GET /api/tasks/ — 获取列表"""
        tasks = [
            {"id": 1, "title": "学习Django"},
            {"id": 2, "title": "搭建API"},
        ]
        return JsonResponse({"data": tasks})
    
    def post(self, request):
        """POST /api/tasks/ — 创建任务"""
        data = json.loads(request.body)
        return JsonResponse({"id": 3, **data}, status=201)

@method_decorator(csrf_exempt, name='dispatch')
class TaskDetailView(View):
    """任务详情视图"""
    
    def get(self, request, pk):
        """GET /api/tasks/5/ — 获取详情"""
        return JsonResponse({"id": pk, "title": f"任务{pk}"})
    
    def put(self, request, pk):
        """PUT /api/tasks/5/ — 更新"""
        data = json.loads(request.body)
        return JsonResponse({"id": pk, **data})
    
    def delete(self, request, pk):
        """DELETE /api/tasks/5/ — 删除"""
        return HttpResponse(status=204)
```

### 5.2 CBV 路由配置

```python
# apps/tasks/urls.py
from django.urls import path
from . import views

app_name = 'tasks'

urlpatterns = [
    # CBV 需要调用 .as_view() 方法
    path('', views.TaskListView.as_view(), name='task-list'),
    path('<int:pk>/', views.TaskDetailView.as_view(), name='task-detail'),
]
```

### 5.3 FBV vs CBV 对比

| 特性 | FBV | CBV |
|------|-----|-----|
| **写法** | 函数 | 类 |
| **HTTP 方法分发** | 手动 if/elif | 自动映射到方法 |
| **代码复用** | 装饰器 | 继承和 Mixin |
| **可读性** | 简单逻辑更清晰 | 复杂逻辑更清晰 |
| **DRF 推荐** | — | ✅（APIView、ViewSet） |

> 💡 后续学习 DRF 时，会使用更强大的 `APIView` 和 `ViewSet`（基于 CBV）。

## 六、装饰器

### 6.1 限制 HTTP 方法

```python
from django.views.decorators.http import (
    require_GET,              # 只允许 GET
    require_POST,             # 只允许 POST
    require_http_methods,     # 指定允许的方法列表
)

@require_GET
def task_list(request):
    return JsonResponse({"data": []})

@require_POST
def create_task(request):
    return JsonResponse({"id": 1}, status=201)

@require_http_methods(["GET", "POST"])
def tasks(request):
    if request.method == "GET":
        return JsonResponse({"data": []})
    else:
        return JsonResponse({"id": 1}, status=201)
```

### 6.2 CSRF 豁免

### 6.3 自定义装饰器

```python
import functools
from django.http import JsonResponse

def api_login_required(view_func):
    """检查 API 请求是否带有有效的认证信息"""
    @functools.wraps(view_func)
    def wrapper(request, *args, **kwargs):
        # 检查 Authorization 请求头
        auth_header = request.headers.get("Authorization")
        if not auth_header:
            return JsonResponse(
                {"error": "未提供认证信息"},
                status=401
            )
        return view_func(request, *args, **kwargs)
    return wrapper

@api_login_required
def protected_view(request):
    return JsonResponse({"message": "这是受保护的资源"})
```

## 七、错误处理

### 7.1 HTTP 状态码

| 状态码 | 含义 | 使用场景 |
|--------|------|---------|
| **200** | OK | 请求成功 |
| **201** | Created | 创建资源成功 |
| **204** | No Content | 删除成功（无返回体） |
| **400** | Bad Request | 请求参数错误 |
| **401** | Unauthorized | 未认证 |
| **403** | Forbidden | 无权限 |
| **404** | Not Found | 资源不存在 |
| **405** | Method Not Allowed | 不支持该 HTTP 方法 |
| **500** | Server Error | 服务器内部错误 |

### 7.2 异常处理

```python
from django.http import JsonResponse, Http404

def task_detail(request, pk):
    try:
        # 从数据库获取任务（后续会用 ORM）
        task = get_task_by_id(pk)
        if task is None:
            return JsonResponse({"error": "任务不存在"}, status=404)
        return JsonResponse(task)
    except ValueError as e:
        return JsonResponse({"error": str(e)}, status=400)
    except Exception as e:
        return JsonResponse({"error": "服务器内部错误"}, status=500)
```

from django.shortcuts import get_object_or_404
from django.http import JsonResponse

# get_object_or_404 — 查询不到时自动返回 404
def task_detail(request, pk):
    task = get_object_or_404(Task, pk=pk)  # 后续学 ORM 后使用

# 自己封装的 JSON 响应工具
def api_response(data=None, message="success", status=200):
    """统一 API 响应格式"""
    response = {
        "code": status,
        "message": message,
    }
    if data is not None:
        response["data"] = data
    return JsonResponse(response, status=status, json_dumps_params={"ensure_ascii": False})

# 使用
def task_list(request):
    tasks = [{"id": 1, "title": "任务1"}]
    return api_response(data=tasks, message="获取成功")

def task_not_found(request, pk):
    return api_response(message=f"任务 {pk} 不存在", status=404)
