## 一、任务定义详解

### 1.1 @shared_task vs @app.task

```python
# @shared_task — 推荐，不绑定到特定 Celery 实例
from celery import shared_task

@shared_task
def send_email(to, subject, body):
    pass

# @app.task — 绑定到特定实例
from config.celery import app

@app.task
def send_email(to, subject, body):
    pass
```

### 1.2 任务参数

```python
@shared_task(
    bind=True,                    # 第一个参数是 self（任务实例）
    name='tasks.send_email',      # 自定义任务名
    max_retries=3,                # 最大重试次数
    default_retry_delay=60,       # 默认重试间隔（秒）
    soft_time_limit=120,          # 软超时（秒），抛 SoftTimeLimitExceeded
    time_limit=180,               # 硬超时（秒），直接终止
    rate_limit='10/m',            # 频率限制：每分钟最多 10 次
    ignore_result=True,           # 不存储结果（提高性能）
    acks_late=True,               # 任务完成后再确认（保证不丢失）
    queue='email',                # 指定队列
)
def send_email(self, to, subject, body):
    pass
```

## 二、任务重试

```python
@shared_task(bind=True, max_retries=5)
def send_notification_email(self, user_id, message):
    """带重试机制的邮件发送"""
    try:
        from django.contrib.auth import get_user_model
        User = get_user_model()
        user = User.objects.get(pk=user_id)
        
        # 模拟发送邮件
        import smtplib
        # ... 发送逻辑 ...
        
        return {'status': 'sent', 'email': user.email}
    
    except User.DoesNotExist:
        # 用户不存在，不重试
        return {'status': 'failed', 'reason': 'user not found'}
    
    except smtplib.SMTPException as exc:
        # SMTP 错误，指数退避重试
        raise self.retry(
            exc=exc,
            countdown=2 ** self.request.retries * 60,  # 1min, 2min, 4min, 8min, 16min
            max_retries=5,
        )
    
    except Exception as exc:
        # 其他错误，固定间隔重试
        raise self.retry(exc=exc, countdown=60)
```

### 2.1 自动重试装饰器

```python
@shared_task(
    autoretry_for=(ConnectionError, TimeoutError),  # 指定异常自动重试
    retry_kwargs={'max_retries': 3},
    retry_backoff=True,           # 指数退避
    retry_backoff_max=600,        # 最大退避时间（秒）
    retry_jitter=True,            # 添加随机抖动（防止雪崩）
)
def call_external_api(url, data):
    """调用外部 API，自动重试连接错误"""
    import requests
    response = requests.post(url, json=data, timeout=30)
    response.raise_for_status()
    return response.json()
```

## 三、任务调用方式

```python
from apps.tasks.tasks import process_report

# 1. delay() — 最简单的异步调用
result = process_report.delay(project_id=1, month='2025-01')

# 2. apply_async() — 完整控制
result = process_report.apply_async(
    args=[],
    kwargs={'project_id': 1, 'month': '2025-01'},
    countdown=30,              # 30 秒后执行
    eta=datetime(2025, 6, 1, 9, 0),  # 指定时间执行
    expires=3600,              # 1 小时后过期（未执行则丢弃）
    queue='reports',           # 指定队列
    priority=5,                # 优先级（0-9，0 最高）
)

# 3. 查询结果
print(result.id)              # 任务 ID
print(result.status)          # PENDING / STARTED / SUCCESS / FAILURE / RETRY
print(result.ready())         # 是否完成
print(result.successful())    # 是否成功
print(result.get(timeout=30)) # 等待并获取结果
print(result.result)          # 结果（如果已完成）
print(result.traceback)       # 异常信息（如果失败）
```

## 四、任务组合（Workflow）

### 4.1 chain — 链式任务

```python
from celery import chain

# 任务按顺序执行，前一个的结果传给下一个
workflow = chain(
    download_file.s(url),          # 1. 下载文件
    process_file.s(),              # 2. 处理文件（接收下载结果）
    upload_result.s(destination),  # 3. 上传结果
)
result = workflow.apply_async()
```

### 4.2 group — 并行任务

```python
from celery import group

# 多个任务并行执行
job = group([
    send_email.s(user_id=1, message='通知1'),
    send_email.s(user_id=2, message='通知2'),
    send_email.s(user_id=3, message='通知3'),
])
result = job.apply_async()

# 获取所有结果
results = result.get(timeout=30)  # [result1, result2, result3]
```

### 4.3 chord — 并行 + 回调

```python
from celery import chord

# 所有任务完成后执行回调
workflow = chord(
    [
        generate_section.s(section='intro'),
        generate_section.s(section='body'),
        generate_section.s(section='conclusion'),
    ],
    merge_report.s()  # 所有 section 生成后合并
)
result = workflow.apply_async()
```

    """任务完成通知"""

    logger.info(f"任务 '{task.title}' 已完成，通知项目所有者")

@shared_task
def generate_project_report(project_id, date_range):
    """生成项目报告（耗时操作）"""
    from apps.projects.models import Project
    from apps.tasks.models import Task
    from django.db.models import Count, Q
    
    project = Project.objects.get(pk=project_id)
    stats = Task.objects.filter(project=project).aggregate(
        total=Count('id'),
        completed=Count('id', filter=Q(status='completed')),
        pending=Count('id', filter=Q(status='pending')),
        in_progress=Count('id', filter=Q(status='in_progress')),
    )

        'project': project.name,
        'date_range': date_range,
        'statistics': stats,
        'completion_rate': f"{stats['completed'] / max(stats['total'], 1) * 100:.1f}%",

### 6.2 在代码中查询任务状态

```python
from celery.result import AsyncResult

def get_task_status(request, task_id):
    """查询异步任务状态的 API"""
    result = AsyncResult(task_id)
    response = {
        'task_id': task_id,
        'status': result.status,
        'ready': result.ready(),
    }
    if result.ready():
        if result.successful():
            response['result'] = result.result
        else:
            response['error'] = str(result.result)
    return JsonResponse(response)
```
