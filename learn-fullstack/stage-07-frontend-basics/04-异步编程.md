# 第 04 节：异步编程

## 一、为什么需要异步？

JavaScript 是 **单线程** 语言，同一时间只能做一件事。如果遇到耗时操作（网络请求、文件读取），同步执行会 **阻塞** 后续代码。

```javascript
// ❌ 同步阻塞（假设场景）
const data = fetchFromServer();  // 等待 3 秒
console.log(data);                // 3 秒后才执行
updateUI();                       // 界面卡死 3 秒

// ✅ 异步非阻塞
fetchFromServer().then(data => {
    console.log(data);            // 数据回来后执行
});
updateUI();                       // 立刻执行，界面不卡
```

## 二、回调函数（了解即可）

```javascript
// 回调地狱（Callback Hell）
getUser(userId, (user) => {
    getProjects(user.id, (projects) => {
        getTasks(projects[0].id, (tasks) => {
            getComments(tasks[0].id, (comments) => {
                console.log(comments);  // 嵌套太深！
            });
        });
    });
});
```

回调嵌套导致代码难以阅读和维护，因此引入了 Promise。

## 三、Promise

### 3.1 基础用法

```javascript
// 创建 Promise
const fetchTask = (id) => {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            if (id > 0) {
                resolve({ id, title: `任务${id}` });  // 成功
            } else {
                reject(new Error('无效的任务 ID'));     // 失败
            }
        }, 1000);
    });
};

// 使用 Promise
fetchTask(1)
    .then(task => {
        console.log('成功:', task);
        return task.id;  // 返回值传给下一个 then
    })
    .then(id => {
        console.log('任务 ID:', id);
    })
    .catch(error => {
        console.error('失败:', error.message);
    })
    .finally(() => {
        console.log('无论成功或失败都执行');
    });
```

### 3.2 Promise 静态方法

```javascript
// Promise.all — 全部成功才成功
const results = await Promise.all([
    fetch('/api/tasks/'),
    fetch('/api/projects/'),
    fetch('/api/tags/'),
]);
// 任何一个失败就进入 catch

// Promise.allSettled — 等待所有完成（不管成功还是失败）
const results = await Promise.allSettled([
    fetch('/api/tasks/'),
    fetch('/api/invalid/'),
]);
// [
//   { status: 'fulfilled', value: Response },
//   { status: 'rejected', reason: Error },
// ]

// Promise.race — 第一个完成的（成功或失败）
const fastest = await Promise.race([
    fetch('/api/server1/tasks/'),
    fetch('/api/server2/tasks/'),
]);

// Promise.any — 第一个成功的
const first = await Promise.any([
    fetch('/api/server1/tasks/'),
    fetch('/api/server2/tasks/'),
]);
```

## 四、async / await（推荐）

`async/await` 是 Promise 的语法糖，让异步代码看起来像同步代码。

### 4.1 基础用法

```javascript
// async 函数自动返回 Promise
async function getTask(id) {
    const response = await fetch(`/api/v1/tasks/${id}/`);
    
    if (!response.ok) {
        throw new Error(`HTTP ${response.status}`);
    }
    
    const data = await response.json();
    return data;
}

// 调用
const task = await getTask(1);
console.log(task.title);
```

### 4.2 错误处理

```javascript
// try/catch
async function fetchTasks() {
    try {
        const response = await fetch('/api/v1/tasks/');
        if (!response.ok) {
            throw new Error(`请求失败: ${response.status}`);
        }
        const data = await response.json();
        return data.results;
    } catch (error) {
        console.error('获取任务失败:', error.message);
        return [];  // 返回默认值
    }
}

// 并行请求
async function loadDashboard() {
    try {
        // ✅ 并行请求（更快）
        const [tasksRes, projectsRes, statsRes] = await Promise.all([
            fetch('/api/v1/tasks/'),
            fetch('/api/v1/projects/'),
            fetch('/api/v1/tasks/statistics/'),
        ]);
        
        const tasks = await tasksRes.json();
        const projects = await projectsRes.json();
        const stats = await statsRes.json();
        
        return { tasks, projects, stats };
    } catch (error) {
        console.error('加载仪表盘失败:', error);
        throw error;
    }
}

// ❌ 串行请求（慢）
async function loadDashboardSlow() {
    const tasks = await fetch('/api/v1/tasks/').then(r => r.json());
    const projects = await fetch('/api/v1/projects/').then(r => r.json()); // 等 tasks 完成后才开始
    const stats = await fetch('/api/v1/tasks/statistics/').then(r => r.json());
}
```

## 五、Fetch API

`fetch` 是浏览器内置的 HTTP 请求 API。

### 5.1 基础用法

```javascript
// GET 请求
const response = await fetch('/api/v1/tasks/');
const data = await response.json();

// 带查询参数
const params = new URLSearchParams({
    status: 'pending',
    page: 1,
    page_size: 10,
});
const response = await fetch(`/api/v1/tasks/?${params}`);

// POST 请求
const response = await fetch('/api/v1/tasks/', {
    method: 'POST',
    headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${token}`,
    },
    body: JSON.stringify({
        title: '新任务',
        priority: 8,
        project: 1,
    }),
});

// PUT / PATCH / DELETE
await fetch(`/api/v1/tasks/${id}/`, {
    method: 'PATCH',
    headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${token}`,
    },
    body: JSON.stringify({ status: 'completed' }),
});

await fetch(`/api/v1/tasks/${id}/`, {
    method: 'DELETE',
    headers: { 'Authorization': `Bearer ${token}` },
});
```

### 5.2 注意事项

```javascript
// fetch 只在网络错误时 reject，HTTP 错误（404、500）不会 reject！
const response = await fetch('/api/v1/tasks/999/');
console.log(response.ok);      // false
console.log(response.status);  // 404
// 需要手动检查 response.ok

// 封装
async function apiFetch(url, options = {}) {
    const response = await fetch(url, {
        headers: {
            'Content-Type': 'application/json',
            ...options.headers,
        },
        ...options,
    });
    
    if (!response.ok) {
        const error = await response.json().catch(() => ({}));
        throw new Error(error.message || `HTTP ${response.status}`);
    }
    
    if (response.status === 204) return null;
    return response.json();
}
```

## 六、实战：TaskFlow API 客户端

```javascript
class TaskFlowAPI {
    constructor(baseUrl, token = null) {
        this.baseUrl = baseUrl;
        this.token = token;
    }
    
    async request(endpoint, options = {}) {
        const url = `${this.baseUrl}${endpoint}`;
        const headers = {
            'Content-Type': 'application/json',
        };
        if (this.token) {
            headers['Authorization'] = `Bearer ${this.token}`;
        }
        
        const response = await fetch(url, { ...options, headers: { ...headers, ...options.headers } });
        
        if (!response.ok) {
            const error = await response.json().catch(() => ({ message: 'Unknown error' }));
            throw new Error(error.message || `HTTP ${response.status}`);
        }
        
        if (response.status === 204) return null;
        return response.json();
    }
    
    // 认证
    async login(email, password) {
        const data = await this.request('/auth/login/', {
            method: 'POST',
            body: JSON.stringify({ email, password }),
        });
        this.token = data.access;
        return data;
    }
    
    // 任务 CRUD
    async getTasks(params = {}) {
        const query = new URLSearchParams(params).toString();
        return this.request(`/tasks/?${query}`);
    }
    
    async getTask(id) {
        return this.request(`/tasks/${id}/`);
    }
    
    async createTask(data) {
        return this.request('/tasks/', {
            method: 'POST',
            body: JSON.stringify(data),
        });
    }
    
    async updateTask(id, data) {
        return this.request(`/tasks/${id}/`, {
            method: 'PATCH',
            body: JSON.stringify(data),
        });
    }
    
    async deleteTask(id) {
        return this.request(`/tasks/${id}/`, { method: 'DELETE' });
    }
}

// 使用
const api = new TaskFlowAPI('http://localhost:8000/api/v1');
await api.login('user@example.com', 'password');
const tasks = await api.getTasks({ status: 'pending', page: 1 });
```
