# 第 04 节：E2E 端到端测试

## 一、什么是 E2E 测试？

端到端（End-to-End）测试模拟真实用户操作，从浏览器层面验证完整流程。

```
单元测试 → 测试单个函数/组件
集成测试 → 测试模块间协作
E2E 测试 → 测试完整用户流程（打开浏览器 → 操作页面 → 验证结果）
```

## 三、编写 E2E 测试

### 3.1 登录流程

```typescript
// e2e/auth.spec.ts
import { test, expect } from '@playwright/test';

test.describe('认证流程', () => {
    test('用户可以登录', async ({ page }) => {
        await page.goto('/login');
        
        // 填写表单
        await page.fill('input[name="email"]', 'test@example.com');
        await page.fill('input[name="password"]', 'testpass123');
        
        // 点击登录
        await page.click('button[type="submit"]');
        
        // 验证跳转到任务页
        await expect(page).toHaveURL('/tasks');
        
        // 验证导航栏显示用户名
        await expect(page.locator('nav')).toContainText('testuser');
    });
    
    test('错误密码显示错误信息', async ({ page }) => {
        await page.goto('/login');
        
        await page.fill('input[name="email"]', 'test@example.com');
        await page.fill('input[name="password"]', 'wrongpassword');
        await page.click('button[type="submit"]');
        
        // 验证错误提示
        await expect(page.locator('.text-red-500, .text-red-600')).toBeVisible();
    });
    
    test('未登录跳转到登录页', async ({ page }) => {
        await page.goto('/tasks');
        await expect(page).toHaveURL(/\/login/);
    });
});
```

### 3.2 任务 CRUD

```typescript
// e2e/tasks.spec.ts
import { test, expect } from '@playwright/test';

// 登录辅助函数
async function login(page: any) {
    await page.goto('/login');
    await page.fill('input[name="email"]', 'test@example.com');
    await page.fill('input[name="password"]', 'testpass123');
    await page.click('button[type="submit"]');
    await page.waitForURL('/tasks');
}

test.describe('任务管理', () => {
    test.beforeEach(async ({ page }) => {
        await login(page);
    });
    
    test('显示任务列表', async ({ page }) => {
        await page.goto('/tasks');
        
        // 验证页面标题
        await expect(page.locator('h1')).toContainText('任务');
        
        // 验证表格存在
        await expect(page.locator('table')).toBeVisible();
    });
    
    test('创建新任务', async ({ page }) => {
        await page.goto('/tasks');
        
        // 点击新建按钮
        await page.click('text=新建任务');
        
        // 填写表单
        await page.fill('input[name="title"]', 'E2E 测试任务');
        await page.fill('textarea[name="description"]', '这是 Playwright 创建的任务');
        
        // 提交
        await page.click('button:has-text("创建")');
        
        // 验证任务出现在列表中
        await expect(page.locator('table')).toContainText('E2E 测试任务');
    });
    
    test('搜索任务', async ({ page }) => {
        await page.goto('/tasks');
        
        // 输入搜索关键词
        await page.fill('input[placeholder*="搜索"]', 'E2E');
        
        // 等待搜索结果
        await page.waitForTimeout(500); // 等待防抖
        
        // 验证结果
        const rows = page.locator('table tbody tr');
        const count = await rows.count();
        
        for (let i = 0; i < count; i++) {
            await expect(rows.nth(i)).toContainText(/E2E/i);
        }
    });
    
    test('查看任务详情', async ({ page }) => {
        await page.goto('/tasks');
        
        // 点击第一个任务
        await page.locator('table tbody tr').first().click();
        
        // 验证跳转到详情页
        await expect(page).toHaveURL(/\/tasks\/\d+/);
        
        // 验证详情页内容
        await expect(page.locator('h1')).not.toBeEmpty();
    });
    
    test('过滤任务状态', async ({ page }) => {
        await page.goto('/tasks');
        
        // 点击"已完成"过滤
        await page.click('text=已完成');
        
        // 验证 URL 更新
        await expect(page).toHaveURL(/status=completed/);
    });
});
```

### 3.3 Page Object 模式

```typescript
// e2e/pages/LoginPage.ts
import { Page, Locator, expect } from '@playwright/test';

export class LoginPage {
    readonly page: Page;
    readonly emailInput: Locator;
    readonly passwordInput: Locator;
    readonly submitButton: Locator;
    readonly errorMessage: Locator;
    
    constructor(page: Page) {
        this.page = page;
        this.emailInput = page.locator('input[name="email"]');
        this.passwordInput = page.locator('input[name="password"]');
        this.submitButton = page.locator('button[type="submit"]');
        this.errorMessage = page.locator('.text-red-500, .text-red-600');
    }
    
    async goto() {
        await this.page.goto('/login');
    }
    
    async login(email: string, password: string) {
        await this.emailInput.fill(email);
        await this.passwordInput.fill(password);
        await this.submitButton.click();
    }
    
    async expectError() {
        await expect(this.errorMessage).toBeVisible();
    }
}

// e2e/pages/TaskListPage.ts
export class TaskListPage {
    readonly page: Page;
    readonly heading: Locator;
    readonly createButton: Locator;
    readonly searchInput: Locator;
    readonly taskTable: Locator;
    
    constructor(page: Page) {
        this.page = page;
        this.heading = page.locator('h1');
        this.createButton = page.locator('text=新建任务');
        this.searchInput = page.locator('input[placeholder*="搜索"]');
        this.taskTable = page.locator('table');
    }
    
    async goto() {
        await this.page.goto('/tasks');
    }
    
    async search(keyword: string) {
        await this.searchInput.fill(keyword);
        await this.page.waitForTimeout(500);
    }
    
    async getTaskCount() {
        return await this.page.locator('table tbody tr').count();
    }
}

// 在测试中使用
test('创建任务', async ({ page }) => {
    const loginPage = new LoginPage(page);
    await loginPage.goto();
    await loginPage.login('test@example.com', 'testpass123');
    
    const taskPage = new TaskListPage(page);
    await taskPage.goto();
    await taskPage.createButton.click();
    // ...
});
```
