## 一、代码分割与懒加载

### 1.1 路由级代码分割

```tsx
// src/App.tsx
import { Suspense, lazy } from 'react';
import { Routes, Route } from 'react-router-dom';
import Spinner from '@/components/Spinner';

// 懒加载页面组件（按路由拆分 bundle）
const TaskListPage = lazy(() => import('@/pages/TaskListPage'));
const TaskDetailPage = lazy(() => import('@/pages/TaskDetailPage'));
const ProjectListPage = lazy(() => import('@/pages/ProjectListPage'));
const SettingsPage = lazy(() => import('@/pages/SettingsPage'));
const LoginPage = lazy(() => import('@/pages/LoginPage'));

function App() {
    return (
        <Suspense fallback={<Spinner fullScreen />}>
            <Routes>
                <Route path="/tasks" element={<TaskListPage />} />
                <Route path="/tasks/:id" element={<TaskDetailPage />} />
                <Route path="/projects" element={<ProjectListPage />} />
                <Route path="/settings" element={<SettingsPage />} />
                <Route path="/login" element={<LoginPage />} />
            </Routes>
        </Suspense>
    );
}
```

## 二、渲染优化

### 2.1 React.memo — 避免不必要的重新渲染

```tsx
import { memo } from 'react';

// 只有当 props 变化时才重新渲染
const TaskCard = memo(function TaskCard({ task, onComplete }: TaskCardProps) {
    console.log('TaskCard render:', task.id);
    return (
        <div className="p-4 bg-white rounded shadow">
            <h3>{task.title}</h3>
            <button onClick={() => onComplete(task.id)}>完成</button>
        </div>
    );
});

// 自定义比较函数（精细控制）
const TaskCard = memo(
    function TaskCard({ task, onComplete }: TaskCardProps) {
        return <div>...</div>;
    },
    (prevProps, nextProps) => {
        // 只在 task 内容变化时重新渲染
        return (
            prevProps.task.id === nextProps.task.id &&
            prevProps.task.title === nextProps.task.title &&
            prevProps.task.status === nextProps.task.status
        );
    }
);
```

### 2.2 useMemo + useCallback 配合 memo

```tsx
function TaskListPage() {
    const [tasks, setTasks] = useState<Task[]>([]);
    const [filter, setFilter] = useState('all');
    
    // ✅ useMemo 缓存过滤结果
    const filteredTasks = useMemo(() => {
        if (filter === 'all') return tasks;
        return tasks.filter((t) => t.status === filter);
    }, [tasks, filter]);
    
    // ✅ useCallback 缓存回调函数
    const handleComplete = useCallback((id: number) => {
        setTasks((prev) => prev.map((t) =>
            t.id === id ? { ...t, status: 'completed' } : t
        ));
    }, []);
    
    return (
        <div>
            <StatusFilter value={filter} onChange={setFilter} />
            {filteredTasks.map((task) => (
                // TaskCard 使用 memo 包裹，handleComplete 引用稳定
                <TaskCard key={task.id} task={task} onComplete={handleComplete} />
            ))}
        </div>
    );
}
```

### 2.3 虚拟列表（长列表优化）

当列表有上百条数据时，只渲染可视区域内的元素。

```bash
npm install @tanstack/react-virtual
```

```tsx
import { useVirtualizer } from '@tanstack/react-virtual';
import { useRef } from 'react';

function VirtualTaskList({ tasks }: { tasks: Task[] }) {
    const parentRef = useRef<HTMLDivElement>(null);
    
    const virtualizer = useVirtualizer({
        count: tasks.length,
        getScrollElement: () => parentRef.current,
        estimateSize: () => 80, // 每项预估高度 80px
        overscan: 5,            // 上下多渲染 5 个
    });
    
    return (
        <div ref={parentRef} className="h-[600px] overflow-auto">
            <div
                style={{ height: `${virtualizer.getTotalSize()}px`, position: 'relative' }}
            >
                {virtualizer.getVirtualItems().map((virtualItem) => {
                    const task = tasks[virtualItem.index];
                    return (
                        <div
                            key={task.id}
                            style={{
                                position: 'absolute',
                                top: 0,
                                transform: `translateY(${virtualItem.start}px)`,
                                width: '100%',
                            }}
                        >
                            <TaskCard task={task} />
                        </div>
                    );
                })}
            </div>
        </div>
    );
}
```

### 4.2 预请求（Prefetch）

```tsx
import { useQueryClient } from '@tanstack/react-query';

function TaskListItem({ task }: { task: Task }) {
    const queryClient = useQueryClient();
    
    // 鼠标悬停时预请求详情
    const handleMouseEnter = () => {
        queryClient.prefetchQuery({
            queryKey: ['tasks', task.id],
            queryFn: () => taskService.getById(task.id).then((r) => r.data),
            staleTime: 60 * 1000,
        });
    };
    
    return (
        <Link to={`/tasks/${task.id}`} onMouseEnter={handleMouseEnter}>
            {task.title}
        </Link>
    );
}
```

```tsx
// src/main.tsx
import { onCLS, onFID, onLCP, onFCP, onTTFB } from 'web-vitals';

function reportWebVitals(metric: any) {
    console.log(metric.name, metric.value);
    // 发送到分析服务
}

onCLS(reportWebVitals);   // Cumulative Layout Shift
onFID(reportWebVitals);   // First Input Delay
onLCP(reportWebVitals);   // Largest Contentful Paint
onFCP(reportWebVitals);   // First Contentful Paint
onTTFB(reportWebVitals);  // Time to First Byte
```

| 指标 | 含义 | 优秀 |
|------|------|------|
| **LCP** | 最大内容绘制 | < 2.5s |
| **FID** | 首次输入延迟 | < 100ms |
| **CLS** | 累积布局偏移 | < 0.1 |
| **FCP** | 首次内容绘制 | < 1.8s |
| **TTFB** | 首字节时间 | < 800ms |
